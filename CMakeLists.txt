cmake_minimum_required(VERSION 3.20)
project(pcenginerecomp VERSION 0.1.0 LANGUAGES CXX)

# PROJECT_IS_TOP_LEVEL starts in CMake 3.21; retain our 3.20 minimum.
set(PCE_TOP_LEVEL OFF)
if(CMAKE_SOURCE_DIR STREQUAL PROJECT_SOURCE_DIR)
  set(PCE_TOP_LEVEL ON)
endif()
option(PCE_BUILD_TOOLS "Build standalone sandbox runners" ${PCE_TOP_LEVEL})

add_library(pce-runtime STATIC runtime/src/runtime.cpp runtime/src/vdc.cpp runtime/src/raster.cpp runtime/src/scsi.cpp runtime/src/toc.cpp runtime/src/joypad.cpp runtime/src/sector.cpp runtime/src/block.cpp runtime/src/interrupt.cpp runtime/src/video.cpp runtime/src/audio.cpp runtime/src/adpcm.cpp runtime/src/cdda.cpp runtime/src/input_script.cpp runtime/src/scanline.cpp runtime/src/dev_server.cpp)
target_include_directories(pce-runtime PUBLIC runtime/include)
target_link_libraries(pce-runtime PUBLIC ${CMAKE_DL_LIBS})
if(WIN32)
  target_link_libraries(pce-runtime PUBLIC ws2_32)
endif()
target_compile_features(pce-runtime PUBLIC cxx_std_20)
set_target_properties(pce-runtime PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(pce-runtime PRIVATE /W4)
else()
  target_compile_options(pce-runtime PRIVATE -Wall -Wextra -Wpedantic)
endif()

set(PCE_MODULE_SOURCE "" CACHE FILEPATH "Locally generated module.cpp; empty builds interpreter-only sandbox")
set(PCE_MODULE_MANIFEST "" CACHE FILEPATH "Optional generated module manifest; defaults to sibling module.json")
include(cmake/PCEConsumer.cmake)
if(PCE_MODULE_SOURCE)
  if(NOT EXISTS "${PCE_MODULE_SOURCE}")
    message(FATAL_ERROR "PCE_MODULE_SOURCE does not exist")
  endif()
  set(PCE_MODULE "${PCE_MODULE_SOURCE}")
else()
  set(PCE_MODULE runtime/src/no_module.cpp)
endif()
if(PCE_BUILD_TOOLS)
  pce_add_consumer(pce-headless MODULE "${PCE_MODULE}" MANIFEST "${PCE_MODULE_MANIFEST}")
endif()

option(PCE_BUILD_DESKTOP "Build optional SDL3 desktop window/input host" OFF)
if(PCE_BUILD_DESKTOP)
  find_package(SDL3 3.2 CONFIG REQUIRED)
  add_library(pce-desktop-host STATIC runtime/src/desktop.cpp)
  target_link_libraries(pce-desktop-host PUBLIC pce-runtime PRIVATE SDL3::SDL3)
  target_compile_features(pce-desktop-host PUBLIC cxx_std_20)
  if(PCE_BUILD_TOOLS)
    pce_add_consumer(pce-desktop MODULE "${PCE_MODULE}" MANIFEST "${PCE_MODULE_MANIFEST}" DESKTOP)
  endif()
endif()
